T = int(input())

for test_case in range(1, T + 1):
    list_answer = []
    N = int(input())
    
    list_input = list(map(int, input().split()))
    
    for i in range(10):
        if i % 2 == 0:
            max_input = max(list_input)
            list_input.remove(max_input)
            list_answer.append(max_input)
        else:
            min_input = min(list_input)
            list_input.remove(min_input)
            list_answer.append(min_input)
    
    print(f'#{test_case}', *list_answer)
