def drive(index, curr, total):
    global answer

    if index == N - 1:
        answer = min(answer, total + arr[curr][0])
        return

    for i in range(1, N):
        if visited[i] == 1:
            continue

        visited[i] = 1
        drive(index + 1, i, total + arr[curr][i])
        visited[i] = 0

T = int(input())

for test_case in range(1, T + 1):
    answer = float('inf')

    N = int(input())
    visited = [0] * N
    arr = []

    for _ in range(N):
        arr.append(list(map(int, input().split())))

    visited[0] = 1
    drive(0, 0, 0)

    print(f'#{test_case} {answer}')
